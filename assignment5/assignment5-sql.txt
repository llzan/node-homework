# Assignment5a - Task 1 - Find the total price of each of the first 5 roders, ordered by order_id

SELECT orders.order_id, 
ROUND(SUM(products.price * line_items.quantity)::numeric, 2) 
AS total_price FROM orders JOIN line_items 
ON orders.order_id = line_items.order_id 
JOIN products 
ON line_items.product_id = products.product_id 
GROUP BY orders.order_id 
ORDER BY orders.order_id ASC 
LIMIT 5;

# Assignment5a - Task 2 - Understanding Subqueries

SELECT c.customer_name, 
    AVG(t.total_price) AS average_order_price 
FROM customers c 
JOIN (SELECT o.customer_id, o.order_id, 
SUM(p.price * li.quantity) 
    AS total_price FROM orders o 
JOIN line_items li 
    ON o.order_id = li.order_id 
JOIN products p ON li.product_id = p.product_id 
GROUP by o.customer_id, o.order_id) 
    AS t ON c.customer_id = t.customer_id 
GROUP BY c.customer_id, c.customer_name 
ORDER BY c.customer_name;

# Assignment5a - Task 4 - Aggregation with Having 

SELECT
    employees.first_name,
    employees.last_name,
    COUNT(orders.order_id) AS order_count
FROM employees
JOIN orders
    ON employees.employee_id = orders.employee_id
GROUP BY
    employees.employee_id,
    employees.first_name,
    employees.last_name
HAVING COUNT(orders.order_id) > 5
ORDER BY employees.last_name;
